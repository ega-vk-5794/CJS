* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD /iwbep/if_mgw_appl_srv_runtime~create_deep_entity.


  DATA lo_message TYPE REF TO /iwbep/if_message_container.

  CASE iv_entity_name.
    WHEN zcl_zega_epda_maplet_i_mpc=>gc_inspectionanswer.
      DATA :lt_inspection TYPE zcl_ega_epda_inspection_api=>tt_maplet_inspect_answer,
            ls_inspection LIKE LINE OF lt_inspection.
      DATA lt_visit_attachment TYPE zcl_ega_epda_inspection_api=>tt_maplet_visit_attach.
      TYPES:
        BEGIN OF ty_maplet_inspect_answer .
      TYPES: answerset     TYPE TABLE OF zcl_zega_epda_maplet_i_mpc=>ts_answerparameter WITH DEFAULT KEY,
             attachmentset TYPE TABLE OF zcl_zega_epda_maplet_i_mpc=>ts_attachment WITH DEFAULT KEY.
             INCLUDE TYPE zcl_zega_epda_maplet_i_mpc=>ts_inspectionanswer.
      TYPES END OF ty_maplet_inspect_answer .
      DATA: ls_data   TYPE ty_maplet_inspect_answer,
            lt_return TYPE bapiret2_t.

      io_data_provider->read_entry_data( IMPORTING es_data = ls_data ).
      ls_inspection-questionid         = ls_data-questionid.
      ls_inspection-inspectionid       = ls_data-inspectionid.
      ls_inspection-questioniddesc     = ls_data-questioniddesc.
      ls_inspection-permitid           = ls_data-permitid.
      ls_inspection-submissiondate     = ls_data-submissiondate.
      ls_inspection-answerid           = ls_data-answerid.
      ls_inspection-checklistid        = ls_data-checklistid.
      ls_inspection-epdaapplicationid  = ls_data-epdaapplicationid.
      ls_inspection-inspectiontype     = ls_data-inspectiontype.
      ls_inspection-inspectiontypedesc = ls_data-inspectiontypedesc.
      ls_inspection-visttype           = ls_data-visttype.
      ls_inspection-visttypedesc       = ls_data-visttypedesc .
*    ls_inspection-inspectiontask     = ls_data-inspectiontask.
      ls_inspection-createdby          = ls_data-createdby.
      ls_inspection-facilityid         = ls_data-facilityid.
      ls_inspection-newinspectionid    = ls_data-newinspectionid.
      ls_inspection-locationnumber     = ls_data-locationnumber.
      ls_inspection-answerset          = ls_data-answerset.
      ls_inspection-attachment         = VALUE #( FOR wa IN ls_data-attachmentset
                                                  ( inspection_id = wa-inspectionid
                                                  file_mime_type = wa-file_mime_type
                                                  file_descr = wa-file_descr
                                                  file_data = wa-file_data ) ).

      APPEND ls_inspection TO lt_inspection.

      NEW zcl_ega_epda_inspection_api( )->update_visit_checklist(
        EXPORTING
          it_inspection_answer = lt_inspection
          it_visit_attachment  = lt_visit_attachment
        IMPORTING
          et_return = lt_return ).

      copy_data_to_ref(
        EXPORTING
          is_data = ls_data
        CHANGING
          cr_data = er_deep_entity
      ).
    WHEN zcl_zega_epda_maplet_i_mpc=>gc_task.
      CLEAR   :ls_inspection.
      REFRESH :lt_inspection,
               lt_return.
      TYPES:
        BEGIN OF ty_maplet_inspect_answer_task .
      TYPES:
        inspectionanswerset TYPE TABLE OF ty_maplet_inspect_answer WITH DEFAULT KEY,
        attachmentset       TYPE TABLE OF zcl_zega_epda_maplet_i_mpc=>ts_attachment WITH DEFAULT KEY.
        INCLUDE TYPE zcl_zega_epda_maplet_i_mpc=>ts_task.
      TYPES END OF ty_maplet_inspect_answer_task .
      DATA: ls_data_task TYPE ty_maplet_inspect_answer_task,
            lt_data_task TYPE TABLE OF ty_maplet_inspect_answer_task.
**Read the data
      io_data_provider->read_entry_data( IMPORTING es_data = ls_data_task ).
*    READ TABLE ls_data_task-inspectionanswerset INTO DATA(ls_data_tn) INDEX 1.
      LOOP AT ls_data_task-inspectionanswerset INTO DATA(ls_data_tn).
        ls_inspection-questionid         = ls_data_tn-questionid.
        ls_inspection-inspectionid       = ls_data_tn-inspectionid.
        ls_inspection-questioniddesc     = ls_data_tn-questioniddesc.
        ls_inspection-permitid           = ls_data_tn-permitid.
        ls_inspection-submissiondate     = ls_data_tn-submissiondate.
        ls_inspection-answerid           = ls_data_tn-answerid.
        ls_inspection-checklistid        = ls_data_tn-checklistid.
        ls_inspection-epdaapplicationid  = ls_data_tn-epdaapplicationid.
        ls_inspection-inspectiontype     = ls_data_tn-inspectiontype.
        ls_inspection-inspectiontypedesc = ls_data_tn-inspectiontypedesc.
        ls_inspection-visttype           = ls_data_tn-visttype.
        ls_inspection-visttypedesc       = ls_data_tn-visttypedesc .
        ls_inspection-createdby          = ls_data_tn-createdby.
        ls_inspection-facilityid         = ls_data_tn-facilityid.
        ls_inspection-newinspectionid    = ls_data_tn-newinspectionid.
        ls_inspection-locationnumber     = ls_data_tn-locationnumber.
        ls_inspection-answerset          = ls_data_tn-answerset.
        ls_inspection-attachment         = VALUE #( FOR wa IN ls_data_tn-attachmentset
                                                    ( inspection_id = wa-inspectionid
                                                    file_mime_type = wa-file_mime_type
                                                    file_descr = wa-file_descr
                                                    file_data = wa-file_data ) ).

        APPEND ls_inspection TO lt_inspection.
      ENDLOOP.


      lt_visit_attachment = VALUE #( FOR wa IN ls_data_task-attachmentset
                                                        ( inspection_id = ls_data_task-inspectionid
                                                          file_mime_type = wa-file_mime_type
                                                          file_descr = wa-file_descr
                                                          file_data = wa-file_data ) ).

*    ls_data_task-attachmentset =
      DATA lv_visit_status TYPE zde_epda_inspection_status.
      NEW zcl_ega_epda_inspection_api( )->update_visit_checklist(
        EXPORTING
          it_inspection_answer = lt_inspection
          it_visit_attachment  = lt_visit_attachment
        IMPORTING
          ev_visit_status = lv_visit_status
          et_return = lt_return ).

* Visit Status

      IF lv_visit_status IS NOT INITIAL.
        DATA lv_gis_status  TYPE zde_epda_gis_inspection_status.
        SELECT SINGLE gis_status FROM zepda_gis_map_st INTO lv_gis_status
          WHERE sap_status = lv_visit_status.
        IF sy-subrc = 0.
          ls_data_task-recommendation = lv_gis_status.

          DATA lt_values_tab      TYPE STANDARD TABLE OF dd07v.
          DATA ls_values_tab      TYPE dd07v.

          REFRESH lt_values_tab[].
          CLEAR ls_values_tab.
          CALL FUNCTION 'GET_DOMAIN_VALUES'
            EXPORTING
              domname         = 'ZDO_EPDA_GIS_INSPECTION_STATUS'
              text            = 'X'
            TABLES
              values_tab      = lt_values_tab
            EXCEPTIONS
              no_values_found = 1
              OTHERS          = 2.
          READ TABLE lt_values_tab INTO ls_values_tab WITH KEY domvalue_l = lv_gis_status.
          IF sy-subrc = 0.
            ls_data_task-recommendationtext = ls_values_tab-ddtext.
          ENDIF.
        ENDIF.
      ENDIF.


* Validation
      DATA lv_inspection_id TYPE zde_epda_inspection.
      lv_inspection_id = ls_inspection-inspectionid.
      IF ls_inspection-answerset[] IS NOT INITIAL.
        NEW zcl_ega_epda_inspection_api( )->inspection_validation(
          EXPORTING
            iv_inspection_id = lv_inspection_id
          IMPORTING
            et_return = DATA(lt_return_valid) ).
        APPEND LINES OF lt_return_valid TO lt_return.
      ENDIF.

      IF lt_return IS NOT INITIAL.
        lo_message = /iwbep/cl_mgw_msg_container=>get_mgw_msg_container( ).
        lo_message->add_messages_from_bapi( it_bapi_messages = lt_return
                                            iv_add_to_response_header = abap_true ).
        RAISE EXCEPTION TYPE /iwbep/cx_mgw_busi_exception
          EXPORTING
            message_container = lo_message
            textid            = /iwbep/cx_mgw_busi_exception=>business_error.
      ENDIF.

      copy_data_to_ref(
        EXPORTING
          is_data = ls_data_task
        CHANGING
          cr_data = er_deep_entity
      ).

    WHEN 'PortUpdate'.
      DATA: BEGIN OF ls_port,
              iv_case       TYPE scmg_ext_key,
              iv_cancel     TYPE flag,
              portobjupdate TYPE zega_cj_epda_port_obj_tt,
              et_messages   TYPE bapiret2_t,
            END OF ls_port.

      io_data_provider->read_entry_data( IMPORTING es_data = ls_port ).

      CALL FUNCTION 'ZEGA_CJ_EPDA_UPDATE_PORT_OBJ'
        EXPORTING
          iv_case     = ls_port-iv_case
          iv_cancel   = ls_port-iv_cancel
          it_objects  = ls_port-portobjupdate
        IMPORTING
          et_messages = ls_port-et_messages.

      copy_data_to_ref(
        EXPORTING
          is_data = ls_port
        CHANGING
          cr_data = er_deep_entity
      ).

  ENDCASE.

ENDMETHOD.

* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD /iwbep/if_mgw_appl_srv_runtime~execute_action.
  DATA: ls_data    TYPE zcl_zega_epda_maplet_i_mpc=>message,
        ls_data_n  TYPE zcl_zega_epda_maplet_i_mpc=>message_n,
        lv_insid   TYPE zde_epda_inspection,
        lt_return  TYPE bapiret2_t,
        lo_message TYPE REF TO /iwbep/if_message_container.

  DATA:lv_nr         TYPE /iwbep/sb_odata_ty_p_int64,
       lv_recnnr     TYPE recnnr,
       lv_ts         TYPE timestamp,
       lv_date       TYPE dats,
       lv_time       TYPE zde_ega_inspect_time,
       lv_opr_status TYPE zde_epda_inspect_operat_status,
       lv_inspection TYPE zde_epda_inspection.

  IF iv_action_name EQ 'DeleteDraft'.

    TRY.
        lv_inspection = it_parameter[ name = 'inspectionId' ]-value.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    NEW zcl_ega_epda_inspection_api( )->delete_inspection_draft( EXPORTING iv_inspection_id = lv_inspection IMPORTING et_return = lt_return ).

    IF lt_return IS NOT INITIAL.
      lo_message = /iwbep/cl_mgw_msg_container=>get_mgw_msg_container( ).
      lo_message->add_messages_from_bapi( it_bapi_messages          = lt_return
                                          iv_add_to_response_header = abap_true ).
      RAISE EXCEPTION TYPE /iwbep/cx_mgw_busi_exception
        EXPORTING
          message_container = lo_message
          textid            = /iwbep/cx_mgw_busi_exception=>business_error.
    ENDIF.

    ls_data-information = 'Application has been removed from draft'.
    copy_data_to_ref(
      EXPORTING
        is_data = ls_data
      CHANGING
        cr_data = er_data
    ).
  ELSEIF iv_action_name EQ 'GetInspectionId'.

    TRY.
        lv_recnnr = it_parameter[ name = 'vendorID' ]-value.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    TRY.
        lv_ts = it_parameter[ name = 'startDate' ]-value.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    TRY.
        lv_opr_status = it_parameter[ name = 'operationalStatus' ]-value.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    cft(
      EXPORTING
        i_t = lv_ts
      IMPORTING
        e_d = lv_date
        e_t = lv_time
    ).

    lv_nr = NEW zcl_ega_epda_inspection_api( )->get_next_id(
                                           EXPORTING
                                             iv_inspection_type = '5'
                                             iv_permit_id       = lv_recnnr
                                             iv_date            = lv_date
                                             iv_time            = lv_time
                                             iv_operat_status   = lv_opr_status
                                             iv_case_type       = 'ZE02'
                                           IMPORTING
                                             et_return          = lt_return
                                           ).
    IF lt_return IS NOT INITIAL.
      lo_message = /iwbep/cl_mgw_msg_container=>get_mgw_msg_container( ).
      lo_message->add_messages_from_bapi( it_bapi_messages          = lt_return
                                          iv_add_to_response_header = abap_true ).
      RAISE EXCEPTION TYPE /iwbep/cx_mgw_busi_exception
        EXPORTING
          message_container = lo_message
          textid            = /iwbep/cx_mgw_busi_exception=>business_error.
    ENDIF.

    ls_data_n-getinspectionid = lv_nr.
    ls_data_n-allowed         = 'true'.
    ls_data_n-message         = 'This application can be inspected'.

    copy_data_to_ref(
      EXPORTING
        is_data = ls_data_n
      CHANGING
        cr_data = er_data
    ).
  ENDIF.

ENDMETHOD.

* | [!CX!] /IWBEP/CX_MGW_TECH_EXCEPTION
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD /iwbep/if_mgw_appl_srv_runtime~get_expanded_entityset.

  IF iv_entity_name EQ zcl_zega_epda_maplet_i_mpc=>gc_checklist.
    TYPES:
      BEGIN OF ty_questiontype,
        itemvalueset TYPE TABLE OF zcl_zega_epda_maplet_i_mpc=>ts_itemvalue WITH DEFAULT KEY.
        INCLUDE TYPE zcl_zega_epda_maplet_i_mpc=>ts_questiontype.
  TYPES END OF ty_questiontype.

    TYPES: tt_questiontype TYPE TABLE OF ty_questiontype WITH DEFAULT KEY.

    TYPES:
      BEGIN OF ty_question,
        questiontypeset TYPE tt_questiontype.
        INCLUDE TYPE zcl_zega_epda_maplet_i_mpc=>ts_questions.
  TYPES END OF ty_question.

    TYPES: tt_question TYPE TABLE OF ty_question WITH DEFAULT KEY.

    TYPES:
      BEGIN OF ty_checklist,
        questionsset TYPE tt_question.
        INCLUDE TYPE zcl_zega_epda_maplet_i_mpc=>ts_checklist.
  TYPES END OF ty_checklist.

    TYPES:
    tt_checklist TYPE TABLE OF ty_checklist WITH DEFAULT KEY.

    DATA: ls_checklist    TYPE ty_checklist,
          ls_question     TYPE ty_question,
          ls_questiontype TYPE ty_questiontype,
          ls_itemvalue    TYPE zcl_zega_epda_maplet_i_mpc=>ts_itemvalue,
          lt_checklist    TYPE tt_checklist,
          lt_question     TYPE tt_question,
          lt_questiontype TYPE tt_questiontype,
          lt_itemvalue    TYPE TABLE OF zcl_zega_epda_maplet_i_mpc=>ts_itemvalue,
          lv_vendortype   TYPE zde_ega_business_type.

    TRY .
        lv_vendortype = it_filter_select_options[ property = 'VendorTypeCode' ]-select_options[ 1 ]-low.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    NEW zcl_ega_epda_inspection_api( )->get_inspection_questions(
      EXPORTING
        im_business_type = lv_vendortype
    IMPORTING
      et_quest_list    = DATA(lt_quest_list)
    ).

    SORT lt_quest_list ASCENDING BY business_type category_id question_id option_id.
    LOOP AT lt_quest_list ASSIGNING FIELD-SYMBOL(<fs_qlist>).
      AT NEW business_type.
      ENDAT.
      AT NEW category_id.
        CLEAR ls_checklist.
        REFRESH lt_question.
        ls_checklist-checklistid    = <fs_qlist>-category_id.
        ls_checklist-checklistname  = <fs_qlist>-category_en.
        ls_checklist-vendortype     = <fs_qlist>-businees_type_description."COND #( WHEN lv_vendortype EQ '001' THEN 'Industrial' ELSE 'Professional' ).
        ls_checklist-vendortypecode = <fs_qlist>-business_type. "lv_vendortype.
        ls_checklist-userid         = sy-uname.
        ls_checklist-avoidinroutine = abap_true.
      ENDAT.
      AT NEW question_id.
        CLEAR ls_question.
        ls_question-questionid      = <fs_qlist>-question_id.
        ls_question-question        = <fs_qlist>-question_en.
        ls_question-checklist       = <fs_qlist>-category_id.
        ls_question-readonly        = <fs_qlist>-readonly.
        ls_question-custom_flag     = <fs_qlist>-custom_flag.
        IF ls_question-custom_flag  = abap_false.
          ls_checklist-avoidinroutine = abap_false.
        ENDIF.
        ls_question-map_to_cust_fld = <fs_qlist>-map_to_cust_fld.
      ENDAT.
      AT NEW option_id.
        CLEAR ls_questiontype.
        ls_questiontype-optionid = <fs_qlist>-option_id.
        ls_questiontype-itemtype = <fs_qlist>-option_type.
        ls_questiontype-itemtypetext = <fs_qlist>-option_type_tx.
        ls_questiontype-itemlabel = <fs_qlist>-option_lable_en.
        ls_questiontype-minrequired = <fs_qlist>-minimum_select.
        ls_questiontype-maxrequired = <fs_qlist>-maximum_select.
        ls_questiontype-searchrequired = <fs_qlist>-searchrequired.
        ls_questiontype-ismarks = <fs_qlist>-ismarksrequired.
      ENDAT.

      AT END OF option_id.
        REFRESH lt_itemvalue[].
        LOOP AT <fs_qlist>-item ASSIGNING FIELD-SYMBOL(<fs_item>).
          CLEAR ls_itemvalue.
          ls_itemvalue-id = <fs_item>-item_id.
          ls_itemvalue-value = COND #( WHEN <fs_item>-item_txt_en IS INITIAL THEN <fs_qlist>-question_en ELSE <fs_item>-item_txt_en ).
          APPEND ls_itemvalue TO lt_itemvalue.
        ENDLOOP.
        ls_questiontype-itemvalueset[] = lt_itemvalue[].
        APPEND ls_questiontype TO lt_questiontype.
      ENDAT.
      AT END OF question_id.
        ls_question-questiontypeset[] = lt_questiontype[].
        APPEND ls_question TO lt_question.
        REFRESH lt_questiontype[].
      ENDAT.
      AT END OF category_id.
        ls_checklist-questionsset[] = lt_question[].
        DESCRIBE TABLE lt_question LINES ls_checklist-totalnumberofquestions.
        APPEND ls_checklist TO lt_checklist.
        REFRESH lt_question[].
      ENDAT.
      AT END OF business_type.
      ENDAT.
    ENDLOOP.

    copy_data_to_ref(
      EXPORTING
        is_data = lt_checklist
      CHANGING
        cr_data = er_entityset
        ).

    APPEND 'CHECKLISTSET' TO et_expanded_tech_clauses.
    APPEND 'QUESTIONSSET/QUESTIONTYPESET/ITEMVALUESET' TO et_expanded_tech_clauses.
    APPEND 'QUESTIONSSET/QUESTIONTYPESET' TO et_expanded_tech_clauses.
    APPEND 'QUESTIONTYPESET/ITEMVALUESET' TO et_expanded_tech_clauses.
    APPEND 'QUESTIONTYPESET' TO et_expanded_tech_clauses.

    APPEND 'DROPDOWNFIELDSET/VALUEHELPSET' TO et_expanded_tech_clauses.
  ELSEIF iv_entity_name EQ 'PortAccommodation' OR iv_entity_name EQ 'PortRoom' OR iv_entity_name EQ 'PortBed'.

    DATA: lv_port           TYPE  zde_ega_fshry_port,
          lv_case           TYPE  scmg_ext_key,
          lv_license        TYPE recnnumber,
          lv_contract       TYPE recnnumber,
          lv_reserved_items TYPE  flag,
          lt_buildings      TYPE zega_cj_epda_port_objects_tt.

    IF me->gt_port_rooms IS INITIAL.
      LOOP AT it_filter_select_options INTO DATA(ls_filter).
        CASE ls_filter-property.
          WHEN 'IvCase'.
            LOOP AT ls_filter-select_options INTO DATA(ls_option).
              lv_case = ls_option-low.
            ENDLOOP.
          WHEN 'IvPort'.
            LOOP AT ls_filter-select_options INTO ls_option.
              lv_port = ls_option-low.
            ENDLOOP.
          WHEN 'IvReservedItems'.
            LOOP AT ls_filter-select_options INTO ls_option.
              lv_reserved_items = ls_option-low.
            ENDLOOP.
          WHEN 'IvLicense'.
            LOOP AT ls_filter-select_options INTO ls_option.
              lv_license = ls_option-low.
            ENDLOOP.
          WHEN 'IvContract'.
            LOOP AT ls_filter-select_options INTO ls_option.
              lv_contract = ls_option-low.
            ENDLOOP.
        ENDCASE.
      ENDLOOP.

      CALL FUNCTION 'ZEGA_CJ_EPDA_PORT_OBJECTS'
        EXPORTING
          iv_port           = lv_port
          iv_case           = lv_case
          iv_license        = lv_license
          iv_contract       = lv_contract
          iv_reserved_items = lv_reserved_items
        IMPORTING
          et_building       = lt_buildings
          et_rooms          = me->gt_port_rooms
          et_beds           = me->gt_port_beds.
    ELSE.

      CASE iv_entity_name.
        WHEN 'PortRoom'.
          lt_buildings[] = me->gt_port_rooms[].

        WHEN 'PortBed'.
          lt_buildings[] = me->gt_port_beds[].
      ENDCASE.
      READ TABLE it_key_tab INTO DATA(ls_key) WITH KEY name = 'ArchObjectId'.
      IF sy-subrc EQ 0.
        DELETE lt_buildings WHERE parent_arch_object_id NE ls_key-value.
      ENDIF.
    ENDIF.

    DELETE lt_buildings WHERE available IS INITIAL.

    copy_data_to_ref(
     EXPORTING
       is_data = lt_buildings
     CHANGING
       cr_data = er_entityset
       ).


    APPEND 'PortAccommodation' TO et_expanded_clauses.
    APPEND 'PortRooms' TO et_expanded_clauses.
    APPEND 'PortBeds' TO et_expanded_clauses.
  ENDIF.

ENDMETHOD.

  METHOD taskset_get_entityset.
**Data Declarations
    DATA lr_inspection_api TYPE REF TO zcl_ega_epda_inspection_api.
    DATA:lv_inspection   TYPE zde_epda_inspection,
         lv_noinbox      TYPE c,

         lt_visit_text  TYPE zcl_ega_epda_inspection_api=>tt_domval,

    TRY .
        lv_inspection = it_filter_select_options[ property = 'inspectionId' ]-select_options[ 1 ]-low.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    TRY .
        DATA(lv_status) = it_filter_select_options[ property = 'inspectionStatus' ]-select_options[ 1 ]-low.
      CATCH cx_sy_itab_line_not_found.
    ENDTRY.

    lv_inspection = |{ CONV string( lv_inspection ) WIDTH = 10 ALPHA = IN }|.

**Get the inspection text
    NEW zcl_ega_epda_inspection_api( )->get_dom_val(
       EXPORTING
        iv_domain = 'ZDO_EPDA_INSPECTION_TYPE'
       RECEIVING
         et_domain = lt_insp_text ).
**Get the Visit Text
    NEW zcl_ega_epda_inspection_api( )->get_dom_val(
        EXPORTING
         iv_domain = 'ZDO_EGA_INSPECT_TYPE'
        RECEIVING
          et_domain = lt_visit_text ).

    NEW zcl_ega_epda_inspection_api( )->get_dom_val(
    EXPORTING
     iv_domain = 'ZDO_EPDA_GIS_INSPECTION_STATUS' "'ZDO_EPDA_INSPECTION_STATUS'
    RECEIVING
      et_domain = lt_status_text ).

    SELECT * FROM zepda_gis_map_st INTO TABLE @DATA(lt_gis_map_status).

    CREATE OBJECT lr_inspection_api.

    DATA: lv_p TYPE boolean, lv_d TYPE boolean, lv_c TYPE boolean.
    IF lv_status IS NOT INITIAL.
      IF lv_status EQ 'DRAFT'.
        lv_d = abap_true.
      ELSEIF lv_status EQ 'COMPLETED'.
        lv_c = abap_true.
      ELSEIF lv_status EQ 'PENDING'.
        lv_p = abap_true.
      ENDIF.
    ELSE.
      lv_p = abap_true.
      lv_d = abap_true.
    ENDIF.

    DATA(lt_data) = lr_inspection_api->get_inspector_inbox( im_login_user = sy-uname iv_pending = lv_p iv_draft = lv_d iv_completed = lv_c ).

    DATA ls_entityset TYPE zcl_zega_epda_maplet_i_mpc=>ts_task.

    LOOP AT lt_data ASSIGNING FIELD-SYMBOL(<ls_data>).

      ls_entityset-vendortype = <ls_data>-business_type.
      ls_entityset-wiid = <ls_data>-wi_id.

      ls_entityset-createddate  = ctt( i_d = <ls_data>-assign_date i_t = <ls_data>-assign_time ).
*      ls_entityset-recommendation     = <ls_data>-pre_inspection_status.
*      ls_entityset-recommendationtext = <ls_data>-pre_inspection_status_tx.
      CLEAR ls_entityset-recommendation.
      CLEAR ls_entityset-recommendationtext.
      READ TABLE lt_gis_map_status INTO DATA(ls_gis_map_status) WITH KEY sap_status = <ls_data>-pre_inspection_status.
      IF sy-subrc = 0.
        ls_entityset-recommendation  = ls_gis_map_status-gis_status.
        READ TABLE lt_status_text INTO DATA(ls_status_text) WITH KEY domvalue = ls_gis_map_status-gis_status.
        IF sy-subrc = 0.
          ls_entityset-recommendationtext  = ls_status_text-ddtext.
        ENDIF.
      ENDIF.
      ls_entityset-username   = <ls_data>-inspected_by .
      ls_entityset-colorcode  = <ls_data>-colorcode .
      ls_entityset-remarks    = <ls_data>-remarks .
      ls_entityset-comments   = <ls_data>-comments.
      ls_entityset-ownerid    = <ls_data>-owner_id .
      ls_entityset-ownername  = <ls_data>-bp_name.
      ls_entityset-companyid  = <ls_data>-comp_bp_id .
      ls_entityset-companyname  = <ls_data>-comp_name_en .
      ls_entityset-companyname_ar  = <ls_data>-comp_name_ar .
      ls_entityset-newinspectionid  = <ls_data>-newinspectionid .
      ls_entityset-visittype            = <ls_data>-inspect_type.
      ls_entityset-visittypedesc        = <ls_data>-inspect_type_tx.
      ls_entityset-inspectiontype       = <ls_data>-inspection_type.
      ls_entityset-inspectiontypedesc   = <ls_data>-inspection_type_tx.
      ls_entityset-licensevalidfrom     = ctt( i_d = <ls_data>-permit_issue_date i_t = '000000').
      ls_entityset-licensevalidto       = ctt( i_d = <ls_data>-permit_expiry_date i_t = '000000').
      ls_entityset-vendorid             = <ls_data>-permit_id .
      ls_entityset-mainactivitytext     = <ls_data>-main_activity_text.
      ls_entityset-xcoordinate          = <ls_data>-location_x.
      ls_entityset-ycoordinate          = <ls_data>-location_y.
      ls_entityset-violation            = <ls_data>-violation.
      ls_entityset-proposedfollowupdate = ctt( i_d = <ls_data>-follow_up_date i_t = '000000').
      ls_entityset-districtid           = <ls_data>-location.
      ls_entityset-departmentheadcomment = <ls_data>-departmentheadcomment.
      ls_entityset-userid               = sy-uname.
      ls_entityset-rejected             = <ls_data>-overwritten.
      ls_entityset-operatstatus         = <ls_data>-operat_status.
      ls_entityset-operatstatustext     = <ls_data>-operat_status_tx.
      ls_entityset-status = COND #( WHEN <ls_data>-draft_req EQ 'X' THEN 'DRAFT'
                                    WHEN <ls_data>-pending_req EQ 'X' THEN 'PENDING'
                                    WHEN <ls_data>-completed_req EQ 'X' THEN 'COMPLETED' ).
      APPEND ls_entityset TO et_entityset.
    ENDLOOP.
  ENDMETHOD.
